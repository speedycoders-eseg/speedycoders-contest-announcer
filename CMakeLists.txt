cmake_minimum_required(VERSION 3.20)
project(contest_announcer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CURL REQUIRED)
find_package(LibXml2 REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(contests src/contests.cpp)
target_include_directories(contests PUBLIC src)
target_link_libraries(contests PUBLIC CURL::libcurl LibXml2::LibXml2 nlohmann_json::nlohmann_json)
target_compile_options(contests PRIVATE -Wall -Wextra -Wpedantic)
add_executable(contest-announcer src/main.cpp)
target_link_libraries(contest-announcer PRIVATE contests)
enable_testing()
add_executable(contest-tests tests/tests.cpp)
target_link_libraries(contest-tests PRIVATE contests)
add_test(NAME contests COMMAND contest-tests)
